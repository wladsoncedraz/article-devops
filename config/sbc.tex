% =====================================================================
%  config/sbc.tex — carrega o modelo oficial da SBC (sbc-template.sty v2017)
%
%  Correção de compatibilidade (o .sty oficial fica intacto): com babel e
%  hyperref, o bloco \AtEndPreamble do modelo executa \makeatletter e
%  \makeatother dentro do gancho do \begin{document} e quebra o babel nas
%  versões atuais do LaTeX ("Undefined control sequence \normalsfcodes").
%  Aqui esse bloco é ignorado durante o carregamento e a mesma redefinição
%  de \@lbibitem (bibliografia sem rótulos, com âncoras do hyperref) é
%  aplicada de forma segura no início do documento.
% =====================================================================
\RequirePackage{etoolbox}
\makeatletter
\let\sbc@AtEndPreamble\AtEndPreamble
\let\AtEndPreamble\@gobble
\usepackage{sbc-template}
\let\AtEndPreamble\sbc@AtEndPreamble
% o modelo só define \section e \subsection; \subsubsection ganha o mesmo
% espaçamento de \subsection
\titlespacing*{\subsubsection}{0pt}{6pt plus 4pt minus 4pt}{0pt plus 4pt minus 4pt}
\AtBeginDocument{%
  \@ifpackageloaded{hyperref}{%
    \def\@lbibitem[#1]#2{%
      \@skiphyperreftrue
      \H@item[%
        \ifx\Hy@raisedlink\@empty
          \hyper@anchorstart{cite.#2\@extra@b@citeb}\hyper@anchorend
        \else
          \Hy@raisedlink{\hyper@anchorstart{cite.#2\@extra@b@citeb}\hyper@anchorend}%
        \fi
        \hfill]%
      \@skiphyperreffalse
      \if@filesw
        \begingroup
          \let\protect\noexpand
          \immediate\write\@auxout{\string\bibcite{#2}{#1}}%
        \endgroup
      \fi
      \ignorespaces}%
  }{}}
\makeatother
